// ============================================= // GOOGLE APPS SCRIPT - Truck Booking Backend // ============================================= function doGet() { return HtmlService.createHtmlOutputFromFile('Index') // Name of the HTML file .setTitle('Truck Booking - Roxas to Batangas') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const result = submitBooking(data); return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ ok: false, message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } /** * Main booking function * Called by both google.script.run and fetch (standalone) */ function submitBooking(data) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); // Normalize time let time = data.time || "1:00 AM"; if (time.includes("AM")) time = "1AM"; if (time.includes("PM")) time = "1PM"; // Try to find the correct sheet const sheet = findOrCreateSheet(ss, data.date, time); // Find next empty row (based on Company Name column - column C) const lastRow = sheet.getLastRow(); let nextRow = lastRow + 1; // Better way: scan for first empty company name if (lastRow >= 1) { const companyValues = sheet.getRange(2, 3, lastRow, 1).getValues(); for (let i = 0; i < companyValues.length; i++) { if (!companyValues[i][0] || companyValues[i][0].toString().trim() === "") { nextRow = i + 2; break; } } } // Auto number const no = nextRow - 1; // Prepare data (force uppercase) const rowData = [ no, (data.type || "10W").toString().toUpperCase(), (data.company || "").toString().toUpperCase().trim(), (data.plate || "").toString().toUpperCase().trim(), (data.contact || "").toString().trim(), (data.remarks || "").toString().toUpperCase().trim(), (data.bookedBy || "").toString().toUpperCase().trim(), data.date, data.time ]; // Write to sheet sheet.getRange(nextRow, 1, 1, 9).setValues([rowData]); return { ok: true, message: `Booked successfully! → Sheet: "${sheet.getName()}" (Row ${nextRow})` }; } catch (err) { return { ok: false, message: "Error: " + err.message }; } } /** * Smart sheet finder + creator */ function findOrCreateSheet(ss, dateStr, time) { const possibleNames = generatePossibleSheetNames(dateStr, time); // Try existing sheets first for (let name of possibleNames) { const sheet = ss.getSheetByName(name); if (sheet) return sheet; } // Create new sheet if none found const newName = possibleNames[0]; // Use the first preferred name const sheet = ss.insertSheet(newName); // Add headers const headers = [ "No.", "Type of Vehicle", "Company Name", "Plate Number", "Contact Number", "Remarks", "BOOKED BY", "DATE", "TIME" ]; sheet.getRange(1, 1, 1, 9).setValues([headers]); sheet.getRange(1, 1, 1, 9).setFontWeight("bold").setBackground("#f3f4f6"); sheet.setFrozenRows(1); return sheet; } /** * Generate possible sheet name variations */ function generatePossibleSheetNames(dateStr, time) { const names = []; const d = new Date(dateStr + "T00:00:00"); if (isNaN(d.getTime())) { // Fallback if date is invalid names.push(`${dateStr}-${time}`); names.push(`${dateStr} ${time}`); names.push(dateStr); return names; } const months = [ "JANUARY", "FEBRUARY", "MARCH", "APRIL", "MAY", "JUNE", "JULY", "AUGUST", "SEPTEMBER", "OCTOBER", "NOVEMBER", "DECEMBER" ]; const month = months[d.getMonth()]; const day = d.getDate(); const year = d.getFullYear(); // Most common formats based on your Excel names.push(`${month} ${day},${year}-${time}`); names.push(`${month} ${day},${year}`); names.push(`${month} ${day}`); names.push(`${month} ${day} ${year}`); names.push(`${dateStr}-${time}`); names.push(`${dateStr}`); names.push(`${month} ${day},${year}-1AM`); names.push(`${month} ${day},${year}-1PM`); return names; }
